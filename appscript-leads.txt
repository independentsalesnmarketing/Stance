// Stance Marketing — Lead Pool Google Apps Script
// Deploy as a standalone Web App at script.google.com
//   Execute as: Me  |  Who has access: Anyone
// After deploying, copy the web app URL into GOOGLE_LEADS_SCRIPT_URL env var.
// ──────────────────────────────────────────────────────────────────────────────

function doPost(e) {
  // Use LockService to prevent race conditions on claim attempts
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var timestamp = Utilities.formatDate(new Date(), "America/New_York", "M/d/yyyy h:mm a") + " EST";
    var formType = data.formType || "unknown";

    if (formType === "leadNotification") {
      sendLeadNotificationEmail(data);
      appendLeadActivityLog(ss, timestamp, data.leadId, "Agent Notified", "Admin", "", "Notification sent to " + data.agentEmail);
    } else if (formType === "leadClaimed") {
      appendLeadToSheet(ss, data, timestamp);
      sendLeadClaimedAdminEmail(data);
      appendLeadActivityLog(ss, timestamp, data.leadId, "Lead Claimed", data.claimedByName, data.claimedByEmail, "Lead claimed by " + data.claimedByName);
    } else if (formType === "leadCreated") {
      appendLeadToSheet(ss, data, timestamp);
      appendLeadActivityLog(ss, timestamp, data.leadId, "Lead Created", "Admin", "", "Lead created for " + (data.leadName || ""));
    } else if (formType === "leadOrderSubmitted") {
      updateLeadOrderNumber(ss, data, timestamp);
      sendLeadOrderSubmittedAdminEmail(data);
      var orderActionLabel = data.previousOrderNumber ? "Order Number Updated" : "Order Number Submitted";
      var orderDetails = data.previousOrderNumber
        ? "Order # changed from " + data.previousOrderNumber + " to " + data.orderNumber
        : "Order # " + data.orderNumber + " submitted by " + (data.agentName || "agent");
      appendLeadActivityLog(ss, timestamp, data.leadId, orderActionLabel, data.agentName || "Agent", data.agentEmail || "", orderDetails);
    }

    lock.releaseLock();
    return ContentService
      .createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    lock.releaseLock();
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── Helper: get or create sheet ─────────────────────────────────────────────

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  if (sheet.getRange(1, 1).getValue() === "") {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    var range = sheet.getRange(1, 1, 1, headers.length);
    range.setBackground("#0f4c3a");
    range.setFontColor("#ffffff");
    range.setFontWeight("bold");
    range.setFontSize(11);
    sheet.setFrozenRows(1);
    for (var i = 1; i <= headers.length; i++) {
      sheet.setColumnWidth(i, 160);
    }
  }
  // Move Leads sheet to first position if it's the Leads sheet
  if (name === "Leads") {
    ss.setActiveSheet(sheet);
    ss.moveActiveSheet(1);
  }
  return sheet;
}

function sectionHeader(label) {
  return '<p style="margin:24px 0 10px;font-size:10px;font-weight:700;color:#94a3b8;text-transform:uppercase;letter-spacing:0.15em;border-bottom:1px solid #f1f5f9;padding-bottom:6px;">' + label + '</p>';

function row(label, value) {
  if (!value || value === "—") return '';
  return '<tr>'
    + '<td style="padding:5px 0;font-size:13px;color:#64748b;width:38%;vertical-align:top;">' + label + '</td>'
    + '<td style="padding:5px 0;font-size:13px;color:#1e293b;font-weight:500;">' + value + '</td>'
    + '</tr>';
}

    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("State", data.leadState)
    + row("Provider", data.leadProvider)
    + row("Product", data.leadProduct)
    + '</table>'
    + '</div>'

    + '<p style="margin:0 0 12px;font-size:13px;color:#64748b;">Full details will be revealed after you claim this lead. Act fast — only one agent can claim each lead.</p>'

    + '<div style="text-align:center;margin:24px 0;">'

  MailApp.sendEmail(data.agentEmail, subject, "", { htmlBody: emailWrap(bodyHtml) });
}

// ── Send admin notification when lead is claimed ────────────────────────────

function sendLeadClaimedAdminEmail(data) {
  var recipient = data.adminEmail || "gamblerspassion@gmail.com";
  var subject = "Stance Lead Claimed — " + (data.leadName || "Unknown") + " (" + (data.leadState || "") + ")";

  var claimedAtFormatted = "";
  if (data.claimedAt) {
    try {
      var d = new Date(data.claimedAt);
      claimedAtFormatted = Utilities.formatDate(d, "America/New_York", "M/d/yyyy 'at' h:mm a") + " EST";
    } catch(e) {
      claimedAtFormatted = data.claimedAt;
    }
  }

  var bodyHtml = '<h2 style="margin:0 0 4px;font-size:20px;font-weight:700;color:#0f172a;">Lead Claimed</h2>'
    + '<p style="margin:0 0 20px;font-size:14px;color:#475569;">A lead has been claimed by an agent.</p>'

    + sectionHeader("Lead Details")
    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("Lead Name", data.leadName)
    + row("Lead State", data.leadState)
    + row("Provider", data.provider)
    + row("Product Selected", data.productSelected)
    + '</table>'

    + sectionHeader("Claimed By")
    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("Agent Name", data.claimedByName)
    + row("Agent Email", data.claimedByEmail)
    + row("Date/Time", claimedAtFormatted)
    + '</table>';

  MailApp.sendEmail(recipient, subject, "", { htmlBody: emailWrap(bodyHtml) });
}

// ── Append lead to Leads sheet ──────────────────────────────────────────────

function appendLeadToSheet(ss, data, timestamp) {
  var headers = [
    "Lead ID", "Created At", "Created By", "Status",
    "Full Name", "Address", "State", "Email", "Phone", "DOB",
    "Provider", "Product Selected", "Preferred Install Date", "Preferred Install Time",
    "Notes", "Eligible Agent IDs", "Notified Agent IDs",
    "Claimed By Agent ID", "Claimed By Agent Name", "Claimed By Agent Email", "Claimed At",
    "Removed By", "Removed At", "Admin Notes"
  ];
  var sheet = getOrCreateSheet(ss, "Leads", headers);

  // Check if lead already exists (update) or insert new
  if (data.leadId) {
    var numRows = sheet.getLastRow() - 1;
    if (numRows > 0) {
      var leadIds = sheet.getRange(2, 1, numRows, 1).getValues();
      for (var i = 0; i < leadIds.length; i++) {
        if (String(leadIds[i][0]) === String(data.leadId)) {
          // Update existing row
          var rowNum = i + 2;
          if (data.claimedByName) {
            sheet.getRange(rowNum, 4).setValue("Claimed");
            sheet.getRange(rowNum, 18).setValue(data.claimedByName || "");
            sheet.getRange(rowNum, 19).setValue(data.claimedByName || "");
            sheet.getRange(rowNum, 20).setValue(data.claimedByEmail || "");
            sheet.getRange(rowNum, 21).setValue(data.claimedAt || timestamp);
          }
          return;
        }
      }
    }
  }

  // Insert new lead
  sheet.appendRow([
    data.leadId || "",
    timestamp,
    "Admin",
    data.status || "Unclaimed",
    data.leadName || data.fullName || "",
    data.address || "",
    data.leadState || data.state || "",
    data.email || "",
    data.phone || "",
    data.dob || "",
    data.provider || data.leadProvider || "",
    data.productSelected || data.leadProduct || "",
    data.preferredInstallDate || "",
    data.preferredInstallTime || "",
    data.notes || "",
    "", "", "", "", "", "", "", "", ""
  ]);
}

// ── Append to Lead Activity Log sheet ───────────────────────────────────────

function appendLeadActivityLog(ss, timestamp, leadId, action, actorName, actorEmail, details) {
  var headers = ["Timestamp", "Lead ID", "Action", "Actor Name", "Actor Email", "Details"];
  var sheet = getOrCreateSheet(ss, "Lead Activity Log", headers);
  sheet.appendRow([timestamp, leadId, action, actorName, actorEmail, details]);
}

// ── Agents sheet management ─────────────────────────────────────────────────

function appendAgentToSheet(ss, data, timestamp) {
  var headers = [

    "Can Receive Leads", "Created At", "Updated At"

    data.agentId || "",
    data.agentName || "",
    data.agentEmail || "",
    data.agentTier || "",
    data.approvedStates || "",
    data.activeStatus || "Active",
    data.directProviderAccess || "No",
    data.canReceiveLeads || "Yes",
    timestamp,
    timestamp
  ]);
}

// ── Lead Claim Tokens sheet ─────────────────────────────────────────────────

  var headers = ["Token", "Lead ID", "Agent ID", "Agent Email", "Created At", "Expires At", "Used At", "Status"];
  var sheet = getOrCreateSheet(ss, "Lead Claim Tokens", headers);
  sheet.appendRow([

// ── Send admin notification when an agent submits the order number ──────────

function sendLeadOrderSubmittedAdminEmail(data) {
  var recipient = data.adminEmail || "gamblerspassion@gmail.com";
  var isUpdate = !!data.previousOrderNumber;
  var subject = (isUpdate ? "Stance Order # Updated — " : "Stance Order # Submitted — ")
    + (data.leadName || "Unknown") + " (" + (data.orderNumber || "") + ")";

  var submittedAtFormatted = "";
  if (data.submittedAt) {
    try {
      var d = new Date(data.submittedAt);
      submittedAtFormatted = Utilities.formatDate(d, "America/New_York", "M/d/yyyy 'at' h:mm a") + " EST";
    } catch (e) {
      submittedAtFormatted = data.submittedAt;
    }
  }

  var bodyHtml = '<h2 style="margin:0 0 4px;font-size:20px;font-weight:700;color:#0f172a;">'
    + (isUpdate ? "Order Number Updated" : "Order Number Submitted")
    + '</h2>'
    + '<p style="margin:0 0 20px;font-size:14px;color:#475569;">'
    + (isUpdate
        ? "An agent updated the order number for a claimed lead."
        : "An agent has submitted the carrier order number for a claimed lead. The lead is now marked Completed.")
    + '</p>'

    + sectionHeader("Order")
    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("Order Number", data.orderNumber)
    + (isUpdate ? row("Previous Order #", data.previousOrderNumber) : "")
    + row("Submitted At", submittedAtFormatted)
    + '</table>'

    + sectionHeader("Lead")
    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("Lead Name", data.leadName)
    + row("State", data.leadState)
    + row("Provider", data.provider)
    + row("Product", data.productSelected)
    + '</table>'

    + sectionHeader("Agent")
    + '<table cellpadding="0" cellspacing="0" width="100%">'
    + row("Agent Name", data.agentName)
    + row("Agent Email", data.agentEmail)
    + '</table>';

  MailApp.sendEmail(recipient, subject, "", { htmlBody: emailWrap(bodyHtml) });
}

// ── Update Order # + Status columns in the Leads sheet ──────────────────────

function updateLeadOrderNumber(ss, data, timestamp) {
  var sheet = ss.getSheetByName("Leads");
  if (!sheet) return;
  var numRows = sheet.getLastRow() - 1;
  if (numRows <= 0) return;
  var leadIds = sheet.getRange(2, 1, numRows, 1).getValues();
  for (var i = 0; i < leadIds.length; i++) {
    if (String(leadIds[i][0]) === String(data.leadId)) {
      var rowNum = i + 2;
      sheet.getRange(rowNum, 4).setValue("Completed");                   // Status
      // Append/overwrite the order number into Admin Notes (column 24)
      var existingNotes = sheet.getRange(rowNum, 24).getValue() || "";
      var stamp = "Order # " + (data.orderNumber || "") + " (" + timestamp + ")";
      sheet.getRange(rowNum, 24).setValue(existingNotes ? (existingNotes + "\n" + stamp) : stamp);
      return;
    }
  }
}
